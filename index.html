apps script code
const SHEET_NAME = 'Data';

function getOrCreateSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow(['Date','Weight','Calories','Steps','Sleep']);
    sheet.getRange(1,1,1,5).setFontWeight('bold');
  }
  return sheet;
}

function doGet(e) {
  const action = e.parameter.action;
  if (action === 'getAll') return getAll();
  return ContentService.createTextOutput(JSON.stringify({error:'Unknown action'}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const data = JSON.parse(e.postData.contents);
  if (data.action === 'upsert') return upsert(data.entry);
  if (data.action === 'delete') return deleteRow(data.date);
  return ContentService.createTextOutput(JSON.stringify({error:'Unknown action'}))
    .setMimeType(ContentService.MimeType.JSON);
}

function getAll() {
  const sheet = getOrCreateSheet();
  const rows = sheet.getDataRange().getValues();
  const entries = rows.slice(1).map(r => ({
    date: r[0], weight: r[1]||null, calories: r[2]||null,
    steps: r[3]||null, sleep: r[4]||null
  })).filter(e => e.date);
  return ContentService.createTextOutput(JSON.stringify({ok:true, entries}))
    .setMimeType(ContentService.MimeType.JSON);
}

function upsert(entry) {
  const sheet = getOrCreateSheet();
  const rows = sheet.getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === String(entry.date)) {
      sheet.getRange(i+1,1,1,5).setValues([[
        entry.date, entry.weight||'', entry.calories||'',
        entry.steps||'', entry.sleep||''
      ]]);
      return ContentService.createTextOutput(JSON.stringify({ok:true,action:'updated'}))
        .setMimeType(ContentService.MimeType.JSON);
    }
  }
  sheet.appendRow([entry.date, entry.weight||'', entry.calories||'', entry.steps||'', entry.sleep||'']);
  return ContentService.createTextOutput(JSON.stringify({ok:true,action:'inserted'}))
    .setMimeType(ContentService.MimeType.JSON);
}

function deleteRow(date) {
  const sheet = getOrCreateSheet();
  const rows = sheet.getDataRange().getValues();
  for (let i = rows.length - 1; i >= 1; i--) {
    if (String(rows[i][0]) === String(date)) {
      sheet.deleteRow(i+1);
      return ContentService.createTextOutput(JSON.stringify({ok:true}))
        .setMimeType(ContentService.MimeType.JSON);
    }
  }
  // Row not found — still return ok so UI doesn't block
  return ContentService.createTextOutput(JSON.stringify({ok:true,note:'not found but ok'}))
    .setMimeType(ContentService.MimeType.JSON);
}
